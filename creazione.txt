CREATE TYPE TIPOPRODOTTO AS ENUM ('Fisico', 'Digitale');

CREATE TABLE utente (
    username VARCHAR(50) PRIMARY KEY,
    mail VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(161) NOT NULL, 
    imm_Profilo VARCHAR(255),
    via VARCHAR(100),
    numero_Civico VARCHAR(10),
    CAP VARCHAR(10),
    citta VARCHAR(100)
);

CREATE TABLE venditore (
    partita_Iva VARCHAR(11) PRIMARY KEY,
    username_Utente VARCHAR(50) UNIQUE NOT NULL REFERENCES utente (username) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE prodotto (
    codice_Prodotto SERIAL PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    costo DECIMAL(10, 2) NOT NULL,
    descrizione TEXT NOT NULL,
    tipo TIPOPRODOTTO NOT NULL,
    quantita INTEGER,
    immagine VARCHAR(255),
    peso DECIMAL(10, 2),
    dimensione INTEGER,
    partita_Iva_Venditore VARCHAR(11) NOT NULL REFERENCES venditore (partita_Iva) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE acquisto (
    codice_Acquisto SERIAL PRIMARY KEY,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    costo DECIMAL(10, 2) NOT NULL,
    username_Utente VARCHAR(50) NOT NULL REFERENCES utente (username) ON UPDATE CASCADE,
    codice_Prodotto INTEGER NOT NULL REFERENCES prodotto (codice_Prodotto) ON UPDATE CASCADE
);

CREATE TABLE segue (
    username_Utente VARCHAR(50) REFERENCES utente (username) ON UPDATE CASCADE ON DELETE CASCADE,
    partita_Iva_Venditore VARCHAR(11) REFERENCES venditore (partita_Iva) ON UPDATE CASCADE ON DELETE CASCADE,
    PRIMARY KEY (username_Utente, partita_Iva_Venditore)
);

CREATE TABLE refresh_token (
    id SERIAL PRIMARY KEY,
    token VARCHAR(512) NOT NULL UNIQUE,
    username_utente VARCHAR(50) REFERENCES utente(username) ON DELETE CASCADE,
    creato_il TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
